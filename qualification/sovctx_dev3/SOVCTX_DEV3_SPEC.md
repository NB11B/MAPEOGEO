\[
\text{Tier 1: Ordinary Pupil} \longrightarrow \text{Tier 2: Olympiad Student} \longrightarrow \text{Tier 3: Teacher Methodology} \longrightarrow \text{Tier 4: University Textbook} \longrightarrow \text{Tier 5: Research Paper}
